## 기본 Redux 패턴 (Vanilla Redux + connect)

### 1. Reducer & Store 만들기

```jsx
// src/reducers/counterReducer.js

// 1. action 생성 함수 정의
export const increment = () => ({ type: "INCREMENT" });
export const decrement = () => ({ type: "DECREMENT" });

// 2. 초기 상태
const initialState = { value: 0 };

// 3. reducer 생성
function counterReducer(state = initialState, action) {
  switch (action.type) {
    case "INCREMENT":
      return { ...state, value: state.value + 1 };
    case "DECREMENT":
      return { ...state, value: state.value - 1 };
    default:
      return state;
  }
}

export default counterReducer;
```

```jsx
// src/store.js

import { createStore, combineReducers } from "redux";
import counterReducer from "./reducers/counterReducer";

const rootReducer = combineReducers({
  counter: counterReducer,
});

const store = createStore(rootReducer);

export default store;
```

### 3. connect로 컴포넌트 연결하기

```jsx
// src/App.js

import React from "react";
import { connect } from "react-redux";
import { increment, decrement } from "./reducers/counterReducer.js";

// 1. mapStateToProps: store 상태를 props로 연결
const mapStateToProps = (state) => ({
  count: state.counter.value,
});

// 2. mapDispatchToProps: action을 props로 연결
const mapDispatchToProps = (dispatch) => ({
  increment: () => dispatch(increment()),
  decrement: () => dispatch(decrement()),
});

// 3. component
function Counter({ count, increment, decrement }) {
  return (
    <div>
      <h1>Counter: {count}</h1>
      <button onClick={increment}>+1</button>
      <button onClick={decrement}>-1</button>
    </div>
  );
}

// 4. connect로 store와 연결
export default connect(mapStateToProps, mapDispatchToProps)(Counter);
```

<details>
  <summary>createAction, handleActions, bindActionCreators을 사용할 경우</summary>

```jsx
// src/reducers/counterReducer.js

import { createAction, handleActions } from "redux-actions";

// 1. createAction으로 action 생성 함수 정의
export const increment = createAction("INCREMENT");
export const decrement = createAction("DECREMENT");

// 2. 초기 상태
const initialState = { value: 0 };

// 3. handleActions로 reducer 생성
const counterReducer = handleActions(
  {
    [increment]: (state) => ({
      ...state,
      value: state.value + 1,
    }),

    [decrement]: (state) => ({
      ...state,
      value: state.value - 1,
    }),
  },
  initialState
);

export default counterReducer;
```

```jsx
// src/App.js

import React from "react";
import { connect } from "react-redux";
import { bindActionCreators } from "redux";
import { increment, decrement } from "./reducers/counterReducer.js";

// 1. mapStateToProps: store 상태를 props로 연결
const mapStateToProps = (state) => ({
  count: state.counter.value,
});

// 2. mapDispatchToProps: bindActionCreators 사용하여 action을 props로 연결
const mapDispatchToProps = (dispatch) =>
  bindActionCreators(
    {
      increment,
      decrement,
    },
    dispatch
  );

// 3. component
function Counter({ count, increment, decrement }) {
  return (
    <div>
      <h1>Counter: {count}</h1>
      <button onClick={increment}>+1</button>
      <button onClick={decrement}>-1</button>
    </div>
  );
}

// 4. connect로 store 연결
export default connect(mapStateToProps, mapDispatchToProps)(Counter);
```

## Modern Redux Toolkit 패턴 (RTK + Hooks)

### 1. Reducer & Store 만들기

```jsx
// src/reducers/counterReducer.js

import { createSlice } from "@reduxjs/toolkit";

const counterSlice = createSlice({
  name: "counter",
  initialState: { value: 0 },
  reducers: {
    increment: (state) => {
      state.value += 1;
    },
    decrement: (state) => {
      state.value -= 1;
    },
  },
});

export const { increment, decrement } = counterSlice.actions;
export default counterSlice.reducer;
```

```jsx
// src/store.js

import { configureStore } from "@reduxjs/toolkit";
import counterReducer from "./reducers/counterReducer";

export const store = configureStore({
  reducer: {
    counter: counterReducer,
  },
});
```

### 3. Hook로 컴포넌트 연결하기

```jsx
// src/App.js

import React from "react";
import { useSelector, useDispatch } from "react-redux";
import { increment, decrement } from "./reducers/counterReducer.js";

function Counter() {
  // store 상태 읽기
  const count = useSelector((state) => state.counter.value);

  // dispatch 사용
  const dispatch = useDispatch();

  return (
    <div>
      <h1>Counter: {count}</h1>
      <button onClick={() => dispatch(increment())}>+1</button>
      <button onClick={() => dispatch(decrement())}>-1</button>
    </div>
  );
}

export default Counter;
```

<details>
  <summary>useActions</summary>

`action 생성 함수 → action을 dispatch하는 함수`로 변환
react-redux에 내장되지 않고 제외된 hook
다음을 복사하여 사용

```jsx
// src/useActions.js

import { bindActionCreators } from "redux";
import { useDispatch } from "react-redux";
import { useMemo } from "react";

export function useActions(actions, deps) {
  const dispatch = useDispatch();
  return useMemo(
    () => {
      if (Array.isArray(actions)) {
        return actions.map((a) => bindActionCreators(a, dispatch));
      }
      return bindActionCreators(actions, dispatch);
    },
    deps ? [dispatch, ...deps] : [dispatch]
  );
}
```

```jsx
// src/App.js

import React from "react";
import { useSelector } from "react-redux";
import { increment, decrement } from "./reducers/counterReducer.js";
import { useActions } from "./useActions";

function Counter() {
  const count = useSelector((state) => state.counter.value);

  // increment(), decrement()가 자동으로 dispatch 포함된 함수로 변함
  const { increment: onIncrement, decrement: onDecrement } = useActions({
    increment,
    decrement,
  });

  return (
    <div>
      <h1>Counter: {count}</h1>
      <button onClick={onIncrement}>+1</button>
      <button onClick={onDecrement}>-1</button>
    </div>
  );
}

export default Counter;
```
